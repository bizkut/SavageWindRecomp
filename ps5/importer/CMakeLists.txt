cmake_minimum_required(VERSION 3.16)
project(savagewind_import LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT NSTOOL_SOURCE)
    message(FATAL_ERROR "NSTOOL_SOURCE must name the patched nstool snapshot")
endif()
add_subdirectory(${NSTOOL_SOURCE} ${CMAKE_BINARY_DIR}/nstool EXCLUDE_FROM_ALL)
target_sources(pietendo PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/sha256_ni.c)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$" OR PLATFORM_NAME STREQUAL "ps5")
    set_source_files_properties(${CMAKE_CURRENT_SOURCE_DIR}/sha256_ni.c TARGET_DIRECTORY pietendo
                                PROPERTIES COMPILE_OPTIONS "-mno-avx;-mno-avx2")
endif()
file(GLOB nstool_sources "${NSTOOL_SOURCE}/src/*.cpp")
list(REMOVE_ITEM nstool_sources "${NSTOOL_SOURCE}/src/main.cpp")
if(PLATFORM_NAME STREQUAL "ps5")
    set(import_entry payload_main.cpp)
else()
    set(import_entry host_main.cpp)
endif()
add_executable(savagewind-import importer.cpp ${import_entry})
set_target_properties(savagewind-import PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
target_include_directories(savagewind-import PRIVATE ${NSTOOL_SOURCE}/src)
if(PLATFORM_NAME STREQUAL "ps5")
    target_compile_definitions(savagewind-import PRIVATE SAVAGEWIND_PS5_PAYLOAD=1)
endif()
target_link_libraries(savagewind-import PRIVATE savagewind-package)
add_library(savagewind-package STATIC package_core.cpp package_stream.cpp ${nstool_sources})
target_include_directories(savagewind-package PRIVATE ${NSTOOL_SOURCE}/src)
target_link_libraries(savagewind-package PUBLIC pietendo toolchain fmt lz4 mbedtls)
if(PLATFORM_NAME STREQUAL "ps5")
    target_compile_definitions(savagewind-package PRIVATE SAVAGEWIND_PS5_PAYLOAD=1)
    add_executable(savagewind-package-probe package_probe.cpp
        ../../runtime/romfs_image.c ../../runtime/romfs_build.c)
    target_compile_definitions(savagewind-package-probe PRIVATE ROMFS_PACKAGE=1)
    target_include_directories(savagewind-package-probe PRIVATE ../../runtime ${CMAKE_BINARY_DIR})
    target_link_libraries(savagewind-package-probe PRIVATE savagewind-package)
endif()
if(NOT PLATFORM_NAME STREQUAL "ps5")
    add_executable(savagewind-exefs exefs_main.cpp)
    target_include_directories(savagewind-exefs PRIVATE ${NSTOOL_SOURCE}/src)
    target_link_libraries(savagewind-exefs PRIVATE savagewind-package)
    add_executable(savagewind-package-test package_test.cpp
        ../../runtime/romfs_image.c ../../runtime/romfs_build.c)
    target_compile_definitions(savagewind-package-test PRIVATE ROMFS_PACKAGE=1)
    target_include_directories(savagewind-package-test PRIVATE ../../runtime)
    target_link_libraries(savagewind-package-test PRIVATE savagewind-package pthread)
    add_executable(savagewind-package-capi-test package_capi_test.cpp)
    target_link_libraries(savagewind-package-capi-test PRIVATE savagewind-package pthread)
endif()
